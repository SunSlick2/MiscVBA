Public Function ShiftRelativeRows_v3(ByVal formula As String, ByVal fromRow As Long, ByVal offset As Long) As String
    Dim result As String
    Dim i As Long, n As Long
    Dim ch As String
    
    result = ""
    n = Len(formula)
    i = 1
    
    Do While i <= n
        ch = Mid(formula, i, 1)
        
        ' Detect a cell reference token (Column letters followed by digits)
        If ch Like "[A-Za-z]" Then
            Dim colPart As String
            colPart = ""
            
            ' 1. Collect full column letters
            Do While i <= n And Mid(formula, i, 1) Like "[A-Za-z]"
                colPart = colPart & Mid(formula, i, 1)
                i = i + 1
            Loop
            
            ' 2. Check if digits follow immediately (or if there's a $ right before the digits)
            Dim isAbsoluteRow As Boolean
            isAbsoluteRow = False
            
            If i <= n And Mid(formula, i, 1) = "$" Then
                isAbsoluteRow = True
                i = i + 1 ' Consume the "$"
            End If
            
            ' 3. Now collect digits (row number)
            If i <= n And Mid(formula, i, 1) Like "[0-9]" Then
                Dim rowPart As String
                rowPart = ""
                
                Do While i <= n And Mid(formula, i, 1) Like "[0-9]"
                    rowPart = rowPart & Mid(formula, i, 1)
                    i = i + 1
                Loop
                
                ' 4. Process the row shift
                Dim origRow As Long
                origRow = CLng(rowPart)
                
                If isAbsoluteRow Then
                    ' It was something like B$7 -> keep the $ and the original row
                    result = result & colPart & "$" & rowPart
                Else
                    ' It was something like B7 or $B7 -> shift the row
                    result = result & colPart & CStr(origRow + offset)
                End If
            Else
                ' It was just text/function name (e.g., "INDEX", "MATCH") without trailing digits
                ' If we consumed a "$" that wasn't followed by a digit, put it back (unlikely in Excel syntax)
                If isAbsoluteRow Then
                    result = result & colPart & "$"
                Else
                    result = result & colPart
                End If
            End If
            
        Else
            ' Not a letter, just append character (handles $, numbers, single quotes, brackets, etc.)
            result = result & ch
            i = i + 1
        End If
    Loop
    
    ShiftRelativeRows_v3 = result
End Function
